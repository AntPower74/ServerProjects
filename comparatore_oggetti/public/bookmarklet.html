strumento per Prezzly

Copia i risultati con un click

Un segnalibro (bookmarklet) universale: gira nel tuo browser, sulla pagina che hai già aperto — Google Lens, Subito, Vinted o Facebook Marketplace, lo riconosce da solo — quindi nessun blocco anti-bot lato server. Un click apre (o riattiva) una scheda di Prezzly e le manda i risultati direttamente: niente copia-incolla.

↑ Trascina "Estrai per Prezzly" nella tua barra dei preferiti

Come si usa

  1. Trascina il segnalibro qui sopra nella barra dei preferiti del browser. Non si vede la barra? Ctrl/Cmd + Maiusc + B la mostra.

  2. Apri i risultati di Google Lens, oppure una ricerca su Subito, Vinted o Facebook Marketplace, come fai di solito.

  3. Clicca il segnalibro "Estrai per Prezzly": si apre (o si riattiva) Prezzly con i risultati già estratti. Se il popup è bloccato o la scheda non risponde, l'HTML resta comunque copiato negli appunti come piano B.

Non riesci a trascinarlo? (mobile, o il browser blocca il drag)

Android (Chrome)

  1. Vai su una pagina qualsiasi di Google, tocca la stella nella barra degli indirizzi (o menu ⋮ → Aggiungi ai preferiti).
  2. Menu ⋮ → Preferiti → tocca i tre puntini sul segnalibro appena creato → Modifica.
  3. Cambia il nome in "Estrai per Prezzly", incolla il codice (pulsante "Copia" qui sotto) al posto dell'URL, salva.

iPhone (Safari)

  1. Vai su una pagina qualsiasi, tocca l'icona Condividi (il quadrato con la freccia) → Aggiungi ai preferiti.
  2. Apri i Preferiti (icona libro aperto) → tocca Modifica → seleziona il segnalibro appena creato.
  3. Cambia il nome in "Estrai per Prezzly", incolla il codice al posto dell'indirizzo, tocca Fine.

Computer (Chrome/Edge/Firefox/Safari)

  1. Crea un segnalibro qualsiasi sulla pagina di Google (stella nella barra indirizzi, o menu → Aggiungi ai preferiti).
  2. Modifica quel segnalibro e sostituisci l'indirizzo con il codice qui sotto (copialo tutto, incluso javascript: all'inizio).
  3. Salva. Ora richiamalo dai preferiti quando sei sui risultati di Lens.
javascript:(function(){
  var O = 'https://prezzly.cupto.it';

  function sito(){
    var hn = location.hostname.replace(/^www\./, '');
    if (hn === 'lens.google.com') return 'lens';
    if (hn.indexOf('subito.it') !== -1) return 'subito';
    if (hn.indexOf('vinted.it') !== -1) return 'vinted';
    if (hn === 'facebook.com' || hn.indexOf('.facebook.com') !== -1) return 'marketplace';
    return 'lens';
  }

  var S = sito();
  var e = document.querySelector('%23search') || document.body;
  var h = e.outerHTML;

  function cec(cb){
    var t = document.createElement('textarea');
    t.value = h; t.style.position = 'fixed'; t.style.left = '-9999px';
    document.body.appendChild(t); t.select();
    try { document.execCommand('copy'); } catch (err) {}
    document.body.removeChild(t);
    if (cb) cb();
  }

  function cop(cb){
    if (navigator.clipboard && navigator.clipboard.writeText) {
      navigator.clipboard.writeText(h).then(cb).catch(function(){ cec(cb); });
    } else {
      cec(cb);
    }
  }

  var w = window.open(O + '/?lens=1', 'prezzlyTab');

  if (!w) {
    cop(function(){
      alert('Popup bloccato dal browser: ho copiato l\'HTML negli appunti, incollalo su Prezzly a mano.');
    });
    return;
  }

  var ok = false;

  function onMsg(ev){
    if (ev.source !== w) return;
    if (ev.data === 'prezzly-pronto') {
      w.postMessage({ site: S, html: h }, O);
    } else if (ev.data === 'prezzly-ricevuto') {
      ok = true;
      window.removeEventListener('message', onMsg);
    }
  }

  window.addEventListener('message', onMsg);

  setTimeout(function(){
    if (ok) return;
    window.removeEventListener('message', onMsg);
    cop(function(){
      alert('Non sono riuscito a collegarmi alla scheda Prezzly: ho copiato l\'HTML, incollalo a mano.');
    });
  }, 6000);
})();

Cosa fa esattamente: riconosce il sito dall'indirizzo della pagina, prende il blocco #search (su Google Lens) o tutta la pagina (su Subito, Vinted, Marketplace), apre o riattiva una scheda su prezzly.cupto.it e le passa quell'HTML con un messaggio diretto tra le due schede del tuo browser — nessun dato passa da server esterni. Se il popup viene bloccato, o Prezzly non risponde entro qualche secondo, l'HTML resta comunque copiato negli appunti come piano B. Su altre pagine copia semplicemente quella pagina, innocuo ma inutile.
